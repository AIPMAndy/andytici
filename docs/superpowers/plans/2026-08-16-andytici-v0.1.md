// Textream/Textream/PacingMonitor.swift
import Foundation

enum PacingStatus { case idle, slow, normal, fast }

final class PacingMonitor {
    private struct WordEvent { let timestamp: Date }
    private var events: [WordEvent] = []
    private let windowSeconds: TimeInterval = 5.0
    private let silenceResetSeconds: TimeInterval = 2.0

    var wordsPerMinute: Double {
        let now = Date()
        let recent = events.filter { now.timeIntervalSince($0.timestamp) <= windowSeconds }
        guard !recent.isEmpty else { return 0 }
        let elapsed = recent.last!.timestamp.timeIntervalSince(recent.first!.timestamp)
        guard elapsed > 0 else { return 0 }
        return Double(recent.count) * 60.0 / elapsed
    }

    var status: PacingStatus {
        let wpm = wordsPerMinute
        if wpm == 0 { return .idle }
        if wpm < 150 { return .slow }
        if wpm > 240 { return .fast }
        return .normal
    }

    func recordWord(at date: Date = Date()) {
        // silence reset
        if let last = events.last, date.timeIntervalSince(last.timestamp) > silenceResetSeconds {
            events.removeAll()
        }
        events.append(WordEvent(timestamp: date))
    }

    func reset() {
        events.removeAll()
    }
}
```

// Textream/Textream/ScriptTag.swift
import Foundation

struct ScriptTagToken: Equatable {
    enum Tag: Character, CaseIterable {
        case emphasis = "🎯"
        case highEnergy = "⚡"
        case pause = "⏸"
        case exclaim = "❗"
        case hint = "💡"
        case climax = "🔥"

        var displayName: String {
            switch self {
            case .emphasis: return "关键词"
            case .highEnergy: return "重点句"
            case .pause: return "停顿"
            case .exclaim: return "感叹"
            case .hint: return "提示"
            case .climax: return "情绪高潮"
            }
        }
    }
    let text: String
    let tag: Tag?
}

enum ScriptTag {
    static let tagSet: Set<Character> = Set(ScriptTagToken.Tag.allCases.map { $0.rawValue })

    static func tokenize(_ source: String) -> [ScriptTagToken] {
        var tokens: [ScriptTagToken] = []
        var buffer = ""
        for ch in source {
            if let matched = ScriptTagToken.Tag(rawValue: ch) {
                if !buffer.isEmpty {
                    tokens.append(ScriptTagToken(text: buffer, tag: nil))
                    buffer = ""
                }
                tokens.append(ScriptTagToken(text: String(ch), tag: matched))
            } else {
                buffer.append(ch)
            }
        }
        if !buffer.isEmpty {
            tokens.append(ScriptTagToken(text: buffer, tag: nil))
        }
        return tokens
    }
}

// Textream/Textream/HookTemplate.swift
import Foundation

struct HookTemplate: Identifiable, Equatable {
    enum Category: String, CaseIterable {
        case painPoint = "痛点型"
        case contrast = "反差型"
        case number = "数字型"
        case suspense = "悬念型"
    }

    let id: UUID
    let category: Category
    let template: String
    let exampleFilled: String

    init(category: Category, template: String, exampleFilled: String) {
        self.id = UUID()
        self.category = category
        self.template = template
        self.exampleFilled = exampleFilled
    }

    func fill(with input: String) -> String {
        template.replacingOccurrences(of: "_____", with: input)
    }

    static let all: [HookTemplate] = [
        // 痛点型
        HookTemplate(category: .painPoint, template: "你是不是也 _____", exampleFilled: "你是不是也经常加班到深夜"),
        HookTemplate(category: .painPoint, template: "为什么 _____ 总是做不好", exampleFilled: "为什么你的视频总是没人看"),

        HookTemplate(category: .contrast, template: "99% 的人都做错了 _____", exampleFilled: "99% 的人都做错了选题"),

# Andy题词

> 中文 macOS 口播特化提词器 · 基于 textream fork

## 简介

Andy题词 是一个面向小红书、抖音、视频号口播创作者的免费 macOS 提词器 app。粘贴脚本、选择平台、按播放，就能像专业主播一样流畅录制。

继承自 [textream](https://github.com/f/textream) 的成熟架构（Speech 框架、Dynamic Island 浮窗、远程投屏），并新增中文口播专属能力：

- 🎙️ **实时语速提示** — 数字直接显示当前字/分钟，绿/黄/红三档
- 🎯 **平台预设** — 抖音、小红书、视频号一键切换字数与节奏目标
- ⚡ **Hook 模板** — 10+ 爆款开场模板一键插入
- 🎭 **表情/语气标记** — 在脚本里插入 `🎯 ⚡ ⏸️ ❗ 💡 🔥`，浮窗渲染不同样式
- 🇨🇳 **中文 ASR 优化** — 默认 `zh-CN` + 口播常用词提权

## 安装

### Homebrew（推荐）

```bash
brew install andy/tici/andytici
```

### 手动下载

[Releases 页面](https://github.com/andy/andytici/releases/latest) 下载 DMG。

要求 macOS 15 Sequoia 或更高版本。Apple Silicon 和 Intel 均支持。

### 首次启动

由于 Andy题词 未经 Mac App Store 分发，macOS 可能在首次打开时拦截。在终端运行：

```bash
xattr -cr /Applications/Andy题词.app
```

然后右键 app → **打开**。

## 使用

1. **选择平台** — 首次启动选择主要发布平台（抖音/小红书/视频号）
2. **粘贴脚本** — 把脚本复制粘贴到编辑器
3. **插入 Hook** — 点击「Hook 模板」按钮选个开场
4. **添加表情标记**（可选）— 在关键词前加 `🎯` 等标记
5. **按播放** — 浮窗从屏幕顶部 Dynamic Island 滑下
6. **开始说话** — 词组实时高亮，语速提示在右上角
7. **录制完毕** — 浮窗自动关闭

## 高级

- **远程投屏** — 设置中开启「远程」，用手机扫码即可在 iPad/手机当提词器
- **隐藏屏幕共享** — 录制时（Zoom、FaceTime、OBS）提词器不会被录进去
- **PPTX 导入** — 拖入 .pptx 文件，自动提取 presenter notes
- **URL Scheme** — `andytici://read?text=...` 直接打开浮窗

## 快捷键

- `⌘P` — 播放 / 暂停
- `⌘.` — 关闭浮窗
- `Esc` — 退出全屏模式

## 与 textream 的关系

Andy题词 fork 自 [f/textream](https://github.com/f/textream)，保留其 MIT 协议与全部核心能力（Dynamic Island 浮窗、Speech 框架、BrowserServer、PPTX 导入、屏幕共享隐藏等）。本项目专注于中文口播体验的差异化增强。
